        const startBtn = document.getElementById('startGameBtn');

function playCutscene(filename, onCompleteCallback) {
    const videoUrl = cutsceneFiles.get(filename);
    const videoElement = document.getElementById('cutscenePlayer');
    const ui = document.getElementById('uiLayer');
    const crosshair = document.getElementById('crosshair');

    if (!videoUrl) {
        console.warn(`Cutscene ${filename} not found! Skipping directly to gameplay...`);
        ui.style.display = 'none';
        crosshair.style.display = 'block';
        onCompleteCallback();
        return;
    }
    
    ui.style.transition = "opacity 1s";
    ui.style.opacity = "0";
    
    setTimeout(() => {
        ui.style.display = 'none';
        videoElement.src = videoUrl;
        videoElement.style.display = 'block';
        videoElement.play();
    }, 1000);

    videoElement.onended = () => {
        videoElement.style.transition = "opacity 2s";
        videoElement.style.opacity = "0";
        setTimeout(() => {
            videoElement.style.display = 'none';
            videoElement.style.opacity = "1"; 
            crosshair.style.display = 'block';
            onCompleteCallback();
        }, 2000);
    };
}

function animate() {
    requestAnimationFrame(animate);
    const time = performance.now();
    const delta = (time - prevTime) / 1000;

    // --- PLAYER MOVEMENT ---
    if (isLocked) {
        velocity.x -= velocity.x * 10.0 * delta;
        velocity.z -= velocity.z * 10.0 * delta;
        
        direction.z = Number(moveForward) - Number(moveBackward);
        direction.x = Number(moveRight) - Number(moveLeft);
        direction.normalize(); 

        const speed = 30.0;
        if (moveForward || moveBackward) velocity.z -= direction.z * speed * delta;
        if (moveLeft || moveRight) velocity.x -= direction.x * speed * delta;

        yawObject.translateX(velocity.x * delta);
        yawObject.translateZ(velocity.z * delta);
        
        // Collision bounds
        if (yawObject.position.x > 4) yawObject.position.x = 4;
        if (yawObject.position.x < -4) yawObject.position.x = -4;
        if (yawObject.position.z < -33) yawObject.position.z = -33; 
    }

    // --- REALITY GLITCH TRIGGER ---
    // Trigger when player gets close to the door (Z < -22)
    if (yawObject.position.z < -22 && !realityShifted && !isGlitching) {
        isGlitching = true;
        glitchStartTime = time;
    }

    // --- REALITY GLITCH EXECUTION ---
    if (isGlitching) {
        const elapsed = time - glitchStartTime;
        
        if (elapsed < 1800) { // Glitch lasts 1.8 seconds
            // 1. Neurological FOV Jitter
            camera.fov = 70 + (Math.random() * 15 - 7.5);
            camera.updateProjectionMatrix();

            // 2. Door spatial instability
            theDoor.position.x = (Math.random() * 0.6 - 0.3);
            theDoor.position.y = 3 + (Math.random() * 0.2 - 0.1);

            // 3. Lighting & Environment Flicker
            ceilingLights.forEach(l => l.intensity = Math.random() * 2);
            neonStrips.forEach(s => s.material.emissiveIntensity = Math.random() * 10);
            
            // Strobing background to simulate reality tearing
            scene.background.setHex(Math.random() > 0.8 ? 0xffffff : 0x000000);
            
        } else {
            // --- THE MEMORY OVERWRITE (Post-Glitch) ---
            isGlitching = false;
            realityShifted = true;

            // Stabilize Camera
            camera.fov = 70;
            camera.updateProjectionMatrix();
            scene.background.setHex(0x050505);

            // "The door was never there." - Replace door with seamless wall
            scene.remove(theDoor);
            falseWall.visible = true; 

            // Permanently shift the memory tone (Unsettling Crimson)
            ceilingLights.forEach(l => {
                l.color.setHex(0xff3333); 
                l.intensity = 0.8;
            });
            neonStrips.forEach(s => {
                s.material.emissive.setHex(0xff0000);
                s.material.emissiveIntensity = 2.0;
            });
            scene.fog.color.setHex(0x1a0000); // Shift fog to dark red
        }
    }

    prevTime = time;
    renderer.render(scene, camera);
}

  /**

--- 

   * Convert a height canvas to a tangent‑space normal map.

--- 

   */

--- 

  _createNormalFromHeight(heightCanvas) {

--- 

    const res = this.res;

--- 

    const canvas = document.createElement('canvas');

--- 

    canvas.width = canvas.height = res;

--- 

    const ctx = canvas.getContext('2d');

--- 

    ctx.drawImage(heightCanvas, 0, 0);

--- 

    const imgData = ctx.getImageData(0, 0, res, res);

--- 

    const height = new Float32Array(res * res);

--- 




--- 

    for (let i = 0; i < res * res; i++) {

--- 

      height[i] = imgData.data[i * 4] / 255;   // use red channel

--- 

    }

--- 




--- 

    const normalCtx = canvas.getContext('2d');

--- 

    const normalImg = normalCtx.createImageData(res, res);

--- 




--- 

    for (let y = 0; y < res; y++) {

--- 

      for (let x = 0; x < res; x++) {

--- 

        const idx = y * res + x;

--- 

        const left   = height[y * res + Math.max(x - 1, 0)];

--- 

        const right  = height[y * res + Math.min(x + 1, res - 1)];

--- 

        const up     = height[Math.max(y - 1, 0) * res + x];

--- 

        const down   = height[Math.min(y + 1, res - 1) * res + x];

--- 




--- 

        const dx = (right - left) * 2;

--- 

        const dy = (down - up) * 2;

--- 




--- 

        const nx = -dx;

--- 

        const ny = -dy;

--- 

        const nz = 1.0;

--- 

        const len = Math.sqrt(nx * nx + ny * ny + nz * nz);

--- 




--- 

        const r = ((nx / len) * 0.5 + 0.5) * 255;

--- 

        const g = ((ny / len) * 0.5 + 0.5) * 255;

--- 

        const b = ((nz / len) * 0.5 + 0.5) * 255;

--- 




--- 

        const pi = idx * 4;

--- 

        normalImg.data[pi]     = r;

--- 

        normalImg.data[pi + 1] = g;

--- 

        normalImg.data[pi + 2] = b;

--- 

        normalImg.data[pi + 3] = 255;

--- 

      }

--- 

    }

--- 




--- 

    normalCtx.putImageData(normalImg, 0, 0);

--- 

    const texture = new THREE.CanvasTexture(canvas);

--- 

    texture.wrapS = texture.wrapT = THREE.RepeatWrapping;

--- 

    return texture;

--- 

  }

/**

--- 

 * Factory that creates a `MeshStandardMaterial` with the generated PBR textures.

--- 

 * All meshes in the environment **must** use this factory – no flat colours, no unlit shaders.

--- 

 */

--- 

export class PBRMaterialFactory {

--- 

  constructor() {

--- 

    this.generator = new PBRTextureGenerator(1024);

--- 

    this.cache = new Map();

--- 

  }

--- 




--- 

  /**

--- 

   * Returns a full `MeshStandardMaterial` (or `MeshPhysicalMaterial` if `clearcoat` is set).

--- 

   * @param {string} type - 'concrete', 'corridorMetal', ...

--- 

   * @param {object} overrides - optional roughness/metalness/clearcoat overrides

--- 

   */

--- 

  getMaterial(type, overrides = {}) {

--- 

    const key = `${type}_${JSON.stringify(overrides)}`;

--- 

    if (this.cache.has(key)) return this.cache.get(key);

--- 




--- 

    const textures = this.generator.getTextureSet(type);

--- 




--- 

    // Apply repeating and filtering

--- 

    Object.values(textures).forEach(tex => {

--- 

      tex.wrapS = tex.wrapT = THREE.RepeatWrapping;

--- 

      tex.repeat.set(2, 2);   // scale to world units

--- 

      tex.minFilter = THREE.LinearMipmapLinearFilter;

--- 

      tex.magFilter = THREE.LinearFilter;

--- 

    });

--- 




--- 

    const materialParams = {

--- 

      map: textures.map,

--- 

      roughnessMap: textures.roughnessMap,

--- 

      metalnessMap: textures.metalnessMap,

--- 

      normalMap: textures.normalMap,

--- 

      roughness: overrides.roughness ?? 1.0,

--- 

      metalness: overrides.metalness ?? 0.0,

--- 

      normalScale: new THREE.Vector2(1.5, 1.5),

--- 

      color: new THREE.Color(1, 1, 1)   // multiply with map

--- 

    };

--- 




--- 

    // For wet surfaces or polished metal, use physical material with clearcoat

--- 

    let material;

--- 

    if (overrides.clearcoat) {

--- 

      material = new THREE.MeshPhysicalMaterial(materialParams);

--- 

      material.clearcoat = overrides.clearcoat;

--- 

      material.clearcoatRoughness = overrides.clearcoatRoughness ?? 0.1;

--- 

    } else {

--- 

      material = new THREE.MeshStandardMaterial(materialParams);

--- 

    }

--- 




--- 

    this.cache.set(key, material);

--- 

    return material;

--- 

  }

}
